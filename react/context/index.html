用法

js
import React, { createContext, useState } from 'react';import ChildComponent from './ChildComponent';
// Create the context with a default value of 1export const CountContext = createContext<[number, React.Dispatch<React.SetStateAction<number>>]>([1, () => {}]);
const App: React.FC = () => {  const [count, setCount] = useState(1);
  return (    <CountContext.Provider value={[count, setCount]}>      <div>        <h1>Count: {count}</h1>        <ChildComponent />      </div>    </CountContext.Provider>  );};
export default App;

import React, { useContext } from 'react';import { CountContext } from './App';
const ChildComponent: React.FC = () => {  const [count, setCount] = useContext(CountContext);
  const increment = () => {    setCount(count + 1);  };
  return (    <div>      <h2>Child Component</h2>      <p>Count in child: {count}</p>      <button onClick={increment}>Increment Count</button>    </div>  );};
export default ChildComponent;

createContext 方法法返回的是一个对象，3 个属性：

  • _currentValue： 保存 context 的值的地方

  • Provider： ContextProvider 类型的 jsx

  • Consumer： ContextConsumer 类型的 jsx

  • react 渲染的时候会把 jsx 编译成 render function，函数执行之后就是 vdom

  • vdom 会经历 reconcile 过程转为 fiber 结构，转完之后一次性 commit，也就是更改 dom。

** Provider、Consumer 都是单独的 jsx 类型,**Provider 类型的 vdom 自然会转为对应的 fiber 节点，在 reconcile 的时候会做单独的处理：
