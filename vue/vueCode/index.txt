  // obj的属性也可能是对象或者数组,这里递归实现非常简单,只需要把observer函数在defineProperty 重新调用一遍即可,

  observer(val)

    enumerable: true,
    configurable: true,
    get() {
      // 读取方法

    set(newVal) {
      // 赋值监听方法

function observer(obj) {
  if (typeof obj !== 'object' || obj == null) {

    defineProperty(obj, key, obj[key])

}
