*

/*
* ## 2.柯里化
* 把接受多个参数的函数变换成接受一个单一参数（最初函数的第一个参数）的函数，
* 并且返回接受余下的参数而且返回结果的新函数的技术。
* */

    return (b) => {
        return (c) => {
            return (d) => {
                return a + b + c + d;
            };
        };
    };
};

console.log('2.柯里化:', result);

    return x + y
}

function curryingAdd(x) {
    return function (y) {
        return x + y
    }
}

console.log('2-2.柯里化:', curryingAdd(1)(2)) // 3
