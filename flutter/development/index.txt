type CredentialResponse struct {
    AppID       string `json:"appId"`
    ChannelName string `json:"channelName"`
    UID         uint32 `json:"uid"`
    Token       string `json:"token"`
    ExpiresAt   int64  `json:"expiresAt"`

func (s *Service) IssueRTCCredential(w http.ResponseWriter, r *http.Request) {
    user := s.requireLogin(r)
    roomID := r.PathValue("roomId")

    // 由服务端判断是否在房间、是否被踢出，以及当前是否有发言权限。
    member, err := s.rooms.RequireActiveMember(r.Context(), roomID, user.ID)
    if err != nil {
        writeError(w, http.StatusForbidden, "cannot join this voice room")
        return
    }

    room, err := s.rooms.Get(r.Context(), roomID)
    if err != nil {
        writeError(w, http.StatusNotFound, "voice room not found")
        return
    }

    uid := rtcUID(user.ID) // 稳定映射，不能接受客户端传入的 uid。
    expiresAt := time.Now().Add(30 * time.Minute).Unix()
    role := rtctokenbuilder.RoleSubscriber
    if member.HasMicrophoneSeat && !member.Muted {
        role = rtctokenbuilder.RolePublisher
    }
    token, err := rtctokenbuilder.BuildTokenWithUid(
        s.agoraAppID, s.agoraAppCertificate, room.RTCChannel, uid, role, uint32(expiresAt),
    )
    if err != nil {
        writeError(w, http.StatusInternalServerError, "failed to issue rtc credential")
        return
    }

    writeJSON(w, http.StatusOK, CredentialResponse{
        AppID: s.agoraAppID, ChannelName: room.RTCChannel, UID: uid,
        Token: token, ExpiresAt: expiresAt,
    })
}

class VoiceRtcService {
  VoiceRtcService(this.api);

  final VoiceRoomApi api;
  late final RtcEngine _engine;
  bool _initialized = false;

  Future<void> join(String roomId) async {
    final credential = await api.fetchRtcCredential(roomId);
    final permission = await Permission.microphone.request();
    if (!permission.isGranted) {
      throw StateError('microphone permission is required');
    }

    if (!_initialized) {
      _engine = createAgoraRtcEngine();
      await _engine.initialize(RtcEngineContext(appId: credential.appId));
      _engine.registerEventHandler(RtcEngineEventHandler(
        onTokenPrivilegeWillExpire: (connection, token) {
          _renewToken(roomId);
        },
        onConnectionStateChanged: (connection, state, reason) {
          // 更新可观察状态，供 UI 展示重连/失败状态。
        },
      ));
      await _engine.enableAudio();
      _initialized = true;
    }

    await _engine.joinChannel(
      token: credential.token,
      channelId: credential.channelName,
      uid: credential.uid,
      options: const ChannelMediaOptions(
        channelProfile: ChannelProfileType.channelProfileCommunication,
        clientRoleType: ClientRoleType.clientRoleBroadcaster,
        publishMicrophoneTrack: true,
        autoSubscribeAudio: true,
      ),
    );
  }

  Future<void> _renewToken(String roomId) async {
    final credential = await api.fetchRtcCredential(roomId);
    await _engine.renewToken(credential.token);
  }

  Future<void> leave() async {
    if (!_initialized) return;
    await _engine.leaveChannel();
  }

  Future<void> dispose() async {
    if (!_initialized) return;
    await _engine.release();
    _initialized = false;
  }
