function useNearViewport(rootMargin = "600px 0px") {
  const ref = useRef<HTMLDivElement>(null)
  const [shouldLoad, setShouldLoad] = useState(false)

  useEffect(() => {
    const node = ref.current
    if (!node || shouldLoad) return

    const observer = new IntersectionObserver(([entry]) => {
      if (entry.isIntersecting) {
        setShouldLoad(true)
        observer.disconnect()
      }
    }, { rootMargin, threshold: 0.01 })

    observer.observe(node)
    return () => observer.disconnect()
  }, [rootMargin, shouldLoad])

  return { ref, shouldLoad }
}

export function GameCardImage({ src, alt }: { src: string; alt: string }) {
  const { ref, shouldLoad } = useNearViewport()

  return (
    <div ref={ref} className="relative aspect-[4/5] overflow-hidden rounded-xl bg-neutral-200">

          alt={alt}
          fill
          loading="eager"
          decoding="async"
          sizes="(max-width: 420px) 25vw, 105px"
          className="object-cover"
        />
      )}
    </div>
  )
